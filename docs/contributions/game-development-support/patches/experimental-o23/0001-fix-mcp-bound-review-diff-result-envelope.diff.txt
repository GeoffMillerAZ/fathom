+// O23: review_diff's composed detail is budgeted as one MCP result: content,
+// structuredContent, suppressed, refresh, and budget are the actual dispatch
+// fields. This is intentionally stricter
+// than BA-4's shared, first-item-preserving *payload target*: one enormous
+// nested test or SARIF entry must not turn a review answer into 745 KB.
+
+import (
+	"encoding/json"
+	"errors"
+	"unicode/utf8"
+)

+type reviewOmission struct {
+	Total    int    `json:"total"`
+	Returned int    `json:"returned"`
+	Unit     string `json:"unit"`
+}
+
+func capReviewDiffEnvelope(envelope map[string]any, limit int) (map[string]any, error) {
+	if limit <= 0 {
+		limit = defaultCharBudget
+	}
+	encoded, err := json.Marshal(envelope)
+	if err != nil {
+		return nil, err
+	}
+	if len(encoded) <= limit {
+		return envelope, nil
+	}
+	raw, ok := envelope["structuredContent"].(json.RawMessage)
+	if !ok {
+		return nil, errors.New("structured review result missing")
+	}
+	var original map[string]any
+	if err := json.Unmarshal(raw, &original); err != nil {
+		return nil, err
+	}
+	statuses, ok := original["dimension_status"].(map[string]any)
+	if !ok || len(statuses) != 8 {
+		return nil, errors.New("eight compact dimension statuses missing")
+	}
+	payload := make(map[string]any, len(original))
+	for k, v := range original {
+		payload[k] = v
+	}
+	omitted := make(map[string]reviewOmission)
+	record := func(path string, value any, returned int) {
+		switch rows := value.(type) {
+		case []any:
+			if len(rows) > returned {
+				prev := omitted[path]
+				omitted[path] = reviewOmission{Total: prev.Total + len(rows), Returned: prev.Returned + returned, Unit: "items"}
+			}
+		default:
+			b, _ := json.Marshal(value)
+			if len(b) > returned {
+				prev := omitted[path]
+				omitted[path] = reviewOmission{Total: prev.Total + len(b), Returned: prev.Returned + returned, Unit: "json_bytes"}
+			}
+		}
+	}
+	// Retain every dimension's name/verdict/count, but remove detailed arrays.
+	if dims, ok := payload["dimensions"].([]any); ok {
+		compact := make([]any, len(dims))
+		for i, item := range dims {
+			d, ok := item.(map[string]any)
+			if !ok {
+				return nil, errors.New("review dimension is not an object")
+			}
+			brief := map[string]any{"findings": []any{}}
+			for _, key := range []string{"name", "checked", "count", "reason"} {
+				if value, exists := d[key]; exists {
+					brief[key] = value
+				}
+			}
+			for key, value := range d {
+				if _, keep := brief[key]; !keep {
+					record("dimensions."+key, value, 0)
+				}
+			}
+			if findings, exists := d["findings"]; exists {
+				record("dimensions.findings", findings, 0)
+			}
+			compact[i] = brief
+		}
+		payload["dimensions"] = compact
+	}
+	if gap, ok := payload["test_impact"].(map[string]any); ok {
+		brief := map[string]any{"affected_tests": []any{}, "test_gap": []any{}}
+		for _, key := range []string{"checked", "reason"} {
+			if value, exists := gap[key]; exists {
+				brief[key] = value
+			}
+		}
+		for key, value := range gap {
+			if key != "checked" && key != "reason" {
+				record("test_impact."+key, value, 0)
+			}
+		}
+		payload["test_impact"] = brief
+	}
+	for _, key := range []string{"receipt_sarif", "receipt_check", "receipt"} {
+		if value, exists := payload[key]; exists {
+			record(key, value, 0)
+			delete(payload, key)
+		}
+	}
+	payload["details_complete"] = false
+	wire := make(map[string]any, len(envelope)+2)
+	for key, value := range envelope {
+		wire[key] = value
+	}
+	refreshDisclosure := func() {
+		payload["elided"] = map[string]any{"fields": omitted, "how_to_get_more": "narrow changed_files; query test_impact for one symbol"}
+	}
+	fit := func() (map[string]any, bool, error) {
+		refreshDisclosure()
+		candidate, err := reviewEnvelopeWithPayload(wire, payload)
+		if err != nil {
+			return nil, false, err
+		}
+		b, err := json.Marshal(candidate)
+		if err != nil {
+			return nil, false, err
+		}
+		return candidate, len(b) <= limit, nil
+	}
+	if out, yes, err := fit(); err != nil || yes {
+		return out, err
+	}
+	// Suppression disclosure is envelope metadata, outside the duplicated
+	// payload. Preserve each service's hidden-row count while bounding its
+	// optional IDs and reasons before trimming further review fields.
+	if suppressed, ok := envelope["suppressed"].(map[string]any); ok {
+		if brief, changed := compactReviewSuppressions(suppressed, 2, 96); changed {
+			wire["suppressed"] = brief
+			if out, yes, err := fit(); err != nil || yes {
+				return out, err
+			}
+		}
+	}
+	if refresh, ok := envelope["refresh"].(map[string]string); ok {
+		brief := make(map[string]string, len(refresh))
+		cut := make(map[string]reviewOmission)
+		for service, note := range refresh {
+			short := reviewPrefix(note, 256)
+			brief[service] = short
+			if short != note {
+				cut[service] = reviewOmission{Total: len(note), Returned: len(short), Unit: "utf8_bytes"}
+				if freshness, ok := payload["freshness"].(map[string]any); ok && freshness[service] == note {
+					freshness[service] = short
+					omitted["freshness."+service] = cut[service]
+				}
+			}
+		}
+		if len(cut) > 0 {
+			wire["refresh"] = brief
+			wire["refresh_elided"] = cut
+			wire["details_complete"] = false
+			if out, yes, err := fit(); err != nil || yes {
+				return out, err
+			}
+		}
+	}
+	if budget, ok := envelope["budget"].(map[string]any); ok {
+		if reason, ok := budget["reason"].(string); ok {
+			if short := reviewPrefix(reason, 256); short != reason {
+				brief := make(map[string]any, len(budget)+2)
+				for key, value := range budget {
+					brief[key] = value
+				}
+				brief["reason"] = short
+				brief["details_complete"] = false
+				brief["reason_elided"] = reviewOmission{Total: len(reason), Returned: len(short), Unit: "utf8_bytes"}
+				wire["budget"] = brief
+				if out, yes, err := fit(); err != nil || yes {
+					return out, err
+				}
+			}
+		}
+	}
+	// Optional views are omitted in fixed order, each with a named count.
+	for _, key := range []string{"layers", "service_ref_unextracted", "as_of_scope", "corpus_partial", "verdict_outcome", "index_commit"} {
+		if value, exists := payload[key]; exists {
+			record(key, value, 0)
+			delete(payload, key)
+			if out, yes, err := fit(); err != nil || yes {
+				return out, err
+			}
+		}
+	}
+	if value, exists := payload["service_ref_hops"]; exists {
+		record("service_ref_hops", value, 0)
+		payload["service_ref_hops"] = []any{}
+		if out, yes, err := fit(); err != nil || yes {
+			return out, err
+		}
+	}
+	// Long reasons are still status claims, so keep a UTF-8 prefix and name
+	// precisely how many source bytes could not be carried.
+	if statuses, ok := payload["dimension_status"].(map[string]any); ok {
+		for name, row := range statuses {
+			status, ok := row.(map[string]any)
+			if !ok {
+				continue
+			}
+			if reason, ok := status["unchecked_reason"].(string); ok {
+				if short := reviewPrefix(reason, 128); short != reason {
+					copyRow := make(map[string]any, len(status))
+					for k, v := range status {
+						copyRow[k] = v
+					}
+					copyRow["unchecked_reason"] = short
+					statuses[name] = copyRow
+					omitted["dimension_status."+name+".unchecked_reason"] = reviewOmission{Total: len(reason), Returned: len(short), Unit: "utf8_bytes"}
+				}
+			}
+		}
+	}
+	if dims, ok := payload["dimensions"].([]any); ok {
+		for _, row := range dims {
+			if dim, ok := row.(map[string]any); ok {
+				if reason, ok := dim["reason"].(string); ok {
+					if short := reviewPrefix(reason, 128); short != reason {
+						dim["reason"] = short
+						prev := omitted["dimensions.reason"]
+						omitted["dimensions.reason"] = reviewOmission{Total: prev.Total + len(reason), Returned: prev.Returned + len(short), Unit: "utf8_bytes"}
+					}
+				}
+			}
+		}
+	}
+	if out, yes, err := fit(); err != nil || yes {
+		return out, err
+	}
+	for _, key := range []string{"summary", "service"} {
+		if value, ok := payload[key].(string); ok {
+			if short := reviewPrefix(value, 256); short != value {
+				payload[key] = short
+				omitted[key] = reviewOmission{Total: len(value), Returned: len(short), Unit: "utf8_bytes"}
+			}
+		}
+	}
+	if paths, ok := payload["changed_files"].([]any); ok {
+		var originalBytes, keptBytes int
+		kept := len(paths)
+		if kept > 4 {
+			kept = 4
+		}
+		for i := range paths {
+			if path, ok := paths[i].(string); ok {
+				originalBytes += len(path)
+				if i < kept {
+					short := reviewPrefix(path, 64)
+					keptBytes += len(short)
+					paths[i] = short
+				}
+			}
+		}
+		if originalBytes > keptBytes {
+			omitted["changed_files.path_bytes"] = reviewOmission{Total: originalBytes, Returned: keptBytes, Unit: "utf8_bytes"}
+		}
+		if len(paths) > kept {
+			omitted["changed_files"] = reviewOmission{Total: len(paths), Returned: kept, Unit: "items"}
+			payload["changed_files"] = paths[:kept]
+		}
+	}
+	if notes, ok := payload["freshness"].(map[string]any); ok {
+		for name, value := range notes {
+			if note, ok := value.(string); ok {
+				if short := reviewPrefix(note, 256); short != note {
+					notes[name] = short
+					omitted["freshness."+name] = reviewOmission{Total: len(note), Returned: len(short), Unit: "utf8_bytes"}
+				}
+			}
+		}
+	}
+	if out, yes, err := fit(); err != nil || yes {
+		return out, err
+	}
+	if dims, ok := payload["dimensions"].([]any); ok {
+		omitted["dimensions"] = reviewOmission{Total: len(dims), Returned: 0, Unit: "items"}
+		payload["dimensions"] = []any{}
+	}
+	if out, yes, err := fit(); err != nil || yes {
+		return out, err
+	}
+	if suppressed, ok := envelope["suppressed"].(map[string]any); ok {
+		if brief, changed := compactReviewSuppressions(suppressed, 0, 0); changed {
+			wire["suppressed"] = brief
+			if out, yes, err := fit(); err != nil || yes {
+				return out, err
+			}
+		}
+	}
+	return nil, errors.New("eight review statuses cannot fit the complete MCP result budget")
+}

+func compactReviewSuppressions(original map[string]any, maxItems, maxBytes int) (map[string]any, bool) {
+	out := make(map[string]any, len(original))
+	changed := false
+	for service, value := range original {
+		row, ok := value.(map[string]any)
+		if !ok {
+			out[service] = value
+			continue
+		}
+		brief := make(map[string]any, len(row)+2)
+		for key, field := range row {
+			brief[key] = field
+		}
+		omitted := map[string]reviewOmission{}
+		for _, key := range []string{"ids", "reasons"} {
+			var items []string
+			switch typed := row[key].(type) {
+			case []string:
+				items = typed
+			case []any:
+				valid := true
+				for _, item := range typed {
+					s, ok := item.(string)
+					if !ok {
+						valid = false
+						break
+					}
+					items = append(items, s)
+				}
+				if !valid {
+					continue
+				}
+			default:
+				continue
+			}
+			kept := len(items)
+			if kept > maxItems {
+				kept = maxItems
+			}
+			prefix := make([]string, kept)
+			var totalBytes, returnedBytes int
+			for i, item := range items {
+				totalBytes += len(item)
+				if i < kept {
+					prefix[i] = reviewPrefix(item, maxBytes)
+					returnedBytes += len(prefix[i])
+				}
+			}
+			if kept != len(items) || totalBytes != returnedBytes {
+				changed = true
+				brief[key] = prefix
+				omitted[key] = reviewOmission{Total: len(items), Returned: kept, Unit: "items"}
+				omitted[key+".utf8_bytes"] = reviewOmission{Total: totalBytes, Returned: returnedBytes, Unit: "utf8_bytes"}
+			}
+		}
+		if len(omitted) > 0 {
+			brief["details_complete"] = false
+			brief["elided"] = omitted
+		}
+		out[service] = brief
+	}
+	return out, changed
+}

+func reviewEnvelopeWithPayload(envelope, payload map[string]any) (map[string]any, error) {
+	b, err := json.Marshal(payload)
+	if err != nil {
+		return nil, err
+	}
+	out := make(map[string]any, len(envelope))
+	for k, v := range envelope {
+		out[k] = v
+	}
+	content, ok := envelope["content"].([]any)
+	if !ok || len(content) == 0 {
+		return nil, errors.New("review text content missing")
+	}
+	block, ok := content[0].(map[string]any)
+	if !ok {
+		return nil, errors.New("review text block invalid")
+	}
+	newBlock := make(map[string]any, len(block))
+	for k, v := range block {
+		newBlock[k] = v
+	}
+	newBlock["text"] = string(b)
+	newContent := append([]any(nil), content...)
+	newContent[0] = newBlock
+	out["content"] = newContent
+	out["structuredContent"] = json.RawMessage(b)
+	return out, nil
+}
+
+func reviewPrefix(s string, maxBytes int) string {
+	if len(s) <= maxBytes {
+		return s
+	}
+	i := 0
+	for i < len(s) {
+		_, size := utf8.DecodeRuneInString(s[i:])
+		if size == 0 || i+size > maxBytes {
+			break
+		}
+		i += size
+	}
+	return s[:i]
+}
